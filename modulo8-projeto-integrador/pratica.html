-- Cria o banco da biblioteca (se ainda não existir) com suporte a acentos
CREATE DATABASE IF NOT EXISTS biblioteca CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE biblioteca;                                              -- seleciona o banco

CREATE TABLE leitores (                                      -- uma tabela para alunos E professores
    id   INT AUTO_INCREMENT PRIMARY KEY,                     -- identificador gerado automaticamente
    nome VARCHAR(100) NOT NULL,                              -- nome obrigatório
    tipo CHAR(1) NOT NULL,                                   -- 'A' = aluno, 'P' = professor
    CHECK (tipo IN ('A', 'P'))                               -- impede qualquer outro valor
);

CREATE TABLE livros (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    titulo      VARCHAR(150) NOT NULL,
    autor       VARCHAR(100) NOT NULL,
    exemplares  INT NOT NULL DEFAULT 1,                      -- quantos exemplares a biblioteca possui
    disponiveis INT NOT NULL DEFAULT 1,                      -- quantos estão na prateleira agora
    CHECK (disponiveis >= 0 AND disponiveis <= exemplares)   -- nunca negativo e nunca maior que o total
);

CREATE TABLE emprestimos (
    id              INT AUTO_INCREMENT PRIMARY KEY,
    leitor_id       INT NOT NULL,                            -- quem pegou (chave estrangeira)
    livro_id        INT NOT NULL,                            -- o que pegou (chave estrangeira)
    data_emprestimo DATE NOT NULL,                           -- dia da retirada
    data_prevista   DATE NOT NULL,                           -- dia limite para devolver
    data_devolucao  DATE NULL,                               -- NULL = ainda não devolvido (empréstimo ativo)
    multa           DECIMAL(10,2) NULL,                      -- preenchida na devolução
    FOREIGN KEY (leitor_id) REFERENCES leitores(id),         -- só aceita leitores que existem
    FOREIGN KEY (livro_id)  REFERENCES livros(id)            -- só aceita livros que existem
);

Sprint 1 · Implemente o domínio (Core)

Crie o arquivo Dominio.cs em Biblioteca.Core (depois você pode separar uma classe por arquivo, com F12 e Ctrl+. → “Mover para arquivo”). Repare em quatro ideias: herança (Aluno/Professor), abstração (membros abstract), polimorfismo (o limite muda conforme o tipo) e regras isoladas no serviço.

namespace Biblioteca.Core;                                       // namespace da camada de domínio

public abstract class Leitor                                     // abstract: não existe "leitor genérico", só aluno ou professor
{
    public int Id { get; set; }                                  // identificador (vem do banco)
    public string Nome { get; set; } = "";                       // nome do leitor
    public abstract int LimiteEmprestimos { get; }               // abstract: cada filha é OBRIGADA a definir o seu limite
    public abstract int PrazoDias { get; }                       // idem para o prazo de devolução
    public override string ToString() => Nome;                   // ComboBox usa ToString() para exibir o item
}

public class Aluno : Leitor                                      // ": Leitor" = herança
{
    public override int LimiteEmprestimos => 3;                  // RF04: aluno pode ter até 3 empréstimos ativos
    public override int PrazoDias => 7;                          // RF05: aluno tem 7 dias
}

public class Professor : Leitor
{
    public override int LimiteEmprestimos => 5;                  // RF04: professor até 5
    public override int PrazoDias => 14;                         // RF05: professor tem 14 dias
}

public class Livro
{
    public int Id { get; set; }
    public string Titulo { get; set; } = "";
    public string Autor { get; set; } = "";
    public int Exemplares { get; set; }                          // total de exemplares da biblioteca
    public int Disponiveis { get; set; }                         // exemplares na prateleira agora
    public bool TemDisponivel => Disponiveis > 0;                // propriedade calculada: true se dá para emprestar
    public override string ToString() => $"{Titulo} ({Autor})";  // texto mostrado nas listas
}

public class Emprestimo
{
    public const decimal MultaPorDia = 1.00m;                    // RF07: constante nomeada (m = tipo decimal)

    public int Id { get; set; }
    public Leitor Leitor { get; set; } = null!;                  // null! diz ao compilador: "será preenchido, confie"
    public Livro Livro { get; set; } = null!;
    public DateTime DataEmprestimo { get; set; }                 // dia da retirada
    public DateTime DataPrevista { get; set; }                   // dia limite de devolução
    public DateTime? DataDevolucao { get; set; }                 // DateTime? aceita null: null = ainda não devolvido

    public bool Ativo => DataDevolucao == null;                  // ativo = ainda não foi devolvido

    public bool EstaAtrasado(DateTime hoje)                      // recebe "hoje" por parâmetro: facilita os testes
    {
        return Ativo && hoje.Date > DataPrevista.Date;           // só está atrasado se ativo e passou do prazo
    }

    public decimal CalcularMulta(DateTime dataReferencia)        // RF07
    {
        DateTime fim = DataDevolucao ?? dataReferencia;          // ?? usa a devolução real; se for null, a data de referência
        int diasAtraso = (fim.Date - DataPrevista.Date).Days;    // subtrair datas gera TimeSpan; .Days = número de dias
        return diasAtraso > 0 ? diasAtraso * MultaPorDia : 0m;   // sem atraso, multa zero
    }
}

public class BibliotecaService                                   // regras de negócio: nada de banco, nada de tela
{
    public Emprestimo Emprestar(Leitor leitor, Livro livro,
                                IReadOnlyList<Emprestimo> ativosDoLeitor, DateTime hoje)
    {
        if (!livro.TemDisponivel)                                // RF03
            throw new InvalidOperationException("Não há exemplares disponíveis deste livro.");

        if (ativosDoLeitor.Count >= leitor.LimiteEmprestimos)    // RF04: polimorfismo, cada tipo tem seu limite
            throw new InvalidOperationException(
                $"O limite de {leitor.LimiteEmprestimos} empréstimos simultâneos foi atingido.");

        if (ativosDoLeitor.Any(e => e.EstaAtrasado(hoje)))       // RF06: Any = "existe algum que...?" (LINQ)
            throw new InvalidOperationException("O leitor possui livro em atraso e não pode fazer novo empréstimo.");

        if (ativosDoLeitor.Any(e => e.Livro.Id == livro.Id))     // RF10: mesmo livro duas vezes
            throw new InvalidOperationException("O leitor já está com um exemplar deste livro.");

        // Não altere livro.Disponiveis aqui. O estoque persistente será baixado pelo DAO
        // dentro da mesma transação que grava o empréstimo. Assim, se o banco falhar,
        // o objeto em memória não fica divergente da fonte de verdade.

        return new Emprestimo                                    // cria o empréstimo já com as datas calculadas
        {
            Leitor = leitor,
            Livro = livro,
            DataEmprestimo = hoje.Date,                          // .Date descarta a hora
            DataPrevista = hoje.Date.AddDays(leitor.PrazoDias)   // RF05: prazo depende do tipo do leitor
        };
    }

    public decimal CalcularMultaDevolucao(Emprestimo emprestimo, DateTime hoje)
    {
        if (!emprestimo.Ativo)                                   // não dá para devolver duas vezes
            throw new InvalidOperationException("Este empréstimo já foi devolvido.");

        // Este método valida e calcula a regra, mas não grava estado persistente.
        // A data de devolução e a reposição do exemplar serão confirmadas pelo DAO
        // dentro da mesma transação.
        return emprestimo.CalcularMulta(hoje);                   // RF07: 0 se estiver no prazo
    }
}

Sprint 1 · Proteja as regras com testes (TDD)

Em Biblioteca.Tests, crie BibliotecaServiceTests.cs. Sugestão de TDD: escreva os testes antes do serviço (eles falham: vermelho), implemente o serviço (verde) e refatore. Rode no Gerenciador de Testes (Ctrl+E, T). Todos precisam passar para a tarefa ser considerada pronta.

using Biblioteca.Core;
using Xunit;

namespace Biblioteca.Tests;

public class BibliotecaServiceTests
{
    private static readonly DateTime Hoje = new(2026, 3, 10);   // data fixa: o teste dá o mesmo resultado sempre (R de FIRST)
    private readonly BibliotecaService servico = new();         // objeto testado; um novo para cada teste (xUnit cria uma classe por teste)

    // ---------- métodos auxiliares para montar os dados (evita repetição) ----------
    private static Livro NovoLivro(int id, int disponiveis) => new()
        { Id = id, Titulo = "Livro " + id, Autor = "Autor", Exemplares = disponiveis, Disponiveis = disponiveis };

    private static Emprestimo Ativo(Leitor leitor, Livro livro, DateTime inicio) => new()
        { Leitor = leitor, Livro = livro, DataEmprestimo = inicio, DataPrevista = inicio.AddDays(leitor.PrazoDias) };

    // ---------- RF03, RF05 ----------
    [Fact]
    public void Emprestar_AlunoComExemplar_CriaEmprestimoDeSeteDias()
    {
        var aluno = new Aluno { Id = 1, Nome = "Ana" };                       // ARRANGE
        var livro = NovoLivro(1, 2);
        var emp = servico.Emprestar(aluno, livro, new List<Emprestimo>(), Hoje); // ACT
        Assert.Equal(Hoje.AddDays(7), emp.DataPrevista);                      // ASSERT: aluno = 7 dias
        Assert.Equal(2, livro.Disponiveis);                                   // serviço não altera estoque persistente
    }

    [Fact]
    public void Emprestar_Professor_CriaEmprestimoDeCatorzeDias()
    {
        var prof = new Professor { Id = 2, Nome = "Carla" };
        var emp = servico.Emprestar(prof, NovoLivro(1, 1), new List<Emprestimo>(), Hoje);
        Assert.Equal(Hoje.AddDays(14), emp.DataPrevista);
    }

    [Fact]
    public void Emprestar_SemExemplarDisponivel_LancaExcecao()
    {
        var aluno = new Aluno { Id = 1 };
        Assert.Throws<InvalidOperationException>(                             // espera a exceção da regra RF03
            () => servico.Emprestar(aluno, NovoLivro(1, 0), new List<Emprestimo>(), Hoje));
    }

    // ---------- RF04 ----------
    [Fact]
    public void Emprestar_AlunoNoLimite_LancaExcecao()
    {
        var aluno = new Aluno { Id = 1 };
        var ativos = new List<Emprestimo>                                     // aluno já tem 3 livros diferentes
        {
            Ativo(aluno, NovoLivro(1, 1), Hoje), Ativo(aluno, NovoLivro(2, 1), Hoje), Ativo(aluno, NovoLivro(3, 1), Hoje)
        };
        Assert.Throws<InvalidOperationException>(
            () => servico.Emprestar(aluno, NovoLivro(4, 1), ativos, Hoje));   // o 4º deve ser recusado
    }

    [Fact]
    public void Emprestar_ProfessorComTresAtivos_Permite()
    {
        var prof = new Professor { Id = 2 };
        var ativos = new List<Emprestimo> { Ativo(prof, NovoLivro(1, 1), Hoje), Ativo(prof, NovoLivro(2, 1), Hoje), Ativo(prof, NovoLivro(3, 1), Hoje) };
        var emp = servico.Emprestar(prof, NovoLivro(4, 1), ativos, Hoje);     // professor tem limite 5: o 4º é permitido
        Assert.NotNull(emp);
    }

    // ---------- RF06 ----------
    [Fact]
    public void Emprestar_LeitorComAtraso_LancaExcecao()
    {
        var aluno = new Aluno { Id = 1 };
        var inicio = new DateTime(2026, 3, 1);                                // prazo de 7 dias vence em 08/03
        var ativos = new List<Emprestimo> { Ativo(aluno, NovoLivro(1, 1), inicio) }; // hoje (10/03) já está atrasado
        Assert.Throws<InvalidOperationException>(
            () => servico.Emprestar(aluno, NovoLivro(2, 1), ativos, Hoje));
    }

    // ---------- RF10 ----------
    [Fact]
    public void Emprestar_MesmoLivroDuasVezes_LancaExcecao()
    {
        var aluno = new Aluno { Id = 1 };
        var livro = NovoLivro(1, 2);                                          // há 2 exemplares, mas é o mesmo livro
        var ativos = new List<Emprestimo> { Ativo(aluno, livro, Hoje) };

    // ---------- RF07 ----------
    [Fact]
    public void CalcularMultaDevolucao_NoPrazo_SemMulta()

Sprint 2 · Acesso a dados (Data)

Em Biblioteca.Data, crie DAOs.cs. Observe: parâmetros em todo SQL (RNF01), a transação que grava o empréstimo e baixa o estoque juntos, e o AND disponiveis > 0 que protege contra o “último exemplar” sendo pego por dois usuários.

💡 Fonte da verdade e concorrênciaO banco é a fonte da verdade para estoque e devoluções. As regras são validadas no serviço, mas mudanças persistentes são confirmadas em transação. Em sistemas multiusuário reais, limites, duplicidades e outras regras críticas também podem exigir restrições adicionais no banco ou estratégias de concorrência.
using Biblioteca.Core;
using MySqlConnector;

namespace Biblioteca.Data;

public static class Conexao                                      // único lugar que conhece a connection string
{
    private const string Texto = "Server=localhost;Port=3306;Database=biblioteca;User ID=root;Password=;";
    public static MySqlConnection Criar() => new MySqlConnection(Texto); // cada operação cria e fecha a sua conexão
}


public class LivroDAO
{
    public List<Livro> Listar(string busca)                      // RF09: busca por parte do título
    {
        var lista = new List<Livro>();
        using var con = Conexao.Criar();                         // using var: fecha ao sair do método
        con.Open();
        using var cmd = new MySqlCommand(
            "SELECT id, titulo, autor, exemplares, disponiveis FROM livros " +
            "WHERE titulo LIKE @busca ORDER BY titulo", con);
        cmd.Parameters.AddWithValue("@busca", "%" + busca + "%"); // parâmetro: nunca concatene o texto digitado no SQL (RNF01)
        using var r = cmd.ExecuteReader();                       // leitor de linhas
        while (r.Read())                                         // uma volta por livro encontrado
        {
            lista.Add(new Livro
            {
                Id = r.GetInt32(r.GetOrdinal("id")),             // GetOrdinal acha a posição da coluna pelo nome
                Titulo = r.GetString(r.GetOrdinal("titulo")),
                Autor = r.GetString(r.GetOrdinal("autor")),
                Exemplares = r.GetInt32(r.GetOrdinal("exemplares")),
                Disponiveis = r.GetInt32(r.GetOrdinal("disponiveis"))
            });
        }
        return lista;
    }

    public void Inserir(Livro livro)                             // RF01
    {
        using var con = Conexao.Criar();
        con.Open();
        using var cmd = new MySqlCommand(
            "INSERT INTO livros (titulo, autor, exemplares, disponiveis) VALUES (@t, @a, @e, @e)", con); // @e usado duas vezes: ao cadastrar, todos estão disponíveis
        cmd.Parameters.AddWithValue("@t", livro.Titulo);
        cmd.Parameters.AddWithValue("@a", livro.Autor);
        cmd.Parameters.AddWithValue("@e", livro.Exemplares);
        cmd.ExecuteNonQuery();                                   // INSERT não devolve linhas
    }
}

public class LeitorDAO
{
    public static Leitor Criar(string tipo)                      // static: usada também pelo EmprestimoDAO
    {
        return tipo == "P" ? new Professor() : new Aluno();      // decide a classe filha conforme a coluna "tipo"
    }

    public List<Leitor> Listar()
    {
        var lista = new List<Leitor>();
        using var con = Conexao.Criar();
        con.Open();
        using var cmd = new MySqlCommand("SELECT id, nome, tipo FROM leitores ORDER BY nome", con);
        using var r = cmd.ExecuteReader();
        while (r.Read())
        {
            Leitor l = Criar(r.GetString(r.GetOrdinal("tipo"))); // Aluno ou Professor (polimorfismo)
            l.Id = r.GetInt32(r.GetOrdinal("id"));
            l.Nome = r.GetString(r.GetOrdinal("nome"));
            lista.Add(l);                                        // a lista guarda "Leitor"; cada item é Aluno ou Professor
        }
        return lista;
    }
}

public class EmprestimoDAO
{
    public List<Emprestimo> ListarAtivos()                       // RF08: só os não devolvidos (data_devolucao IS NULL)
    {
        var lista = new List<Emprestimo>();
        using var con = Conexao.Criar();
        con.Open();
        using var cmd = new MySqlCommand(
            "SELECT e.id, e.data_emprestimo, e.data_prevista, " +
            "       l.id AS leitor_id, l.nome AS leitor_nome, l.tipo, " +
            "       v.id AS livro_id, v.titulo, v.autor, v.exemplares, v.disponiveis " +
            "FROM emprestimos e " +
            "INNER JOIN leitores l ON l.id = e.leitor_id " +     // junta com o leitor
            "INNER JOIN livros v ON v.id = e.livro_id " +        // junta com o livro
            "WHERE e.data_devolucao IS NULL " +                  // IS NULL (e não "= NULL") testa valor ausente
            "ORDER BY e.data_prevista", con);
        using var r = cmd.ExecuteReader();
        while (r.Read())
        {
            Leitor leitor = LeitorDAO.Criar(r.GetString(r.GetOrdinal("tipo")));
            leitor.Id = r.GetInt32(r.GetOrdinal("leitor_id"));
            leitor.Nome = r.GetString(r.GetOrdinal("leitor_nome"));
            var livro = new Livro
            {
                Id = r.GetInt32(r.GetOrdinal("livro_id")),
                Titulo = r.GetString(r.GetOrdinal("titulo")),
                Autor = r.GetString(r.GetOrdinal("autor")),
                Exemplares = r.GetInt32(r.GetOrdinal("exemplares")),
                Disponiveis = r.GetInt32(r.GetOrdinal("disponiveis"))
            };
            lista.Add(new Emprestimo                             // monta o objeto completo, com leitor e livro dentro (composição)
            {
                Id = r.GetInt32(r.GetOrdinal("id")),
                Leitor = leitor,
                Livro = livro,
                DataEmprestimo = r.GetDateTime(r.GetOrdinal("data_emprestimo")),
                DataPrevista = r.GetDateTime(r.GetOrdinal("data_prevista"))
            });
        }
        return lista;
    }

    public void Registrar(Emprestimo e)                          // grava o empréstimo E baixa o estoque, juntos
    {
        using var con = Conexao.Criar();
        con.Open();
        using var tx = con.BeginTransaction();                   // inicia a transação: tudo vale ou nada vale
        try
        {
            using var inserir = new MySqlCommand(
                "INSERT INTO emprestimos (leitor_id, livro_id, data_emprestimo, data_prevista) " +
                "VALUES (@l, @v, @de, @dp)", con, tx);           // tx: este comando participa da transação
            inserir.Parameters.AddWithValue("@l", e.Leitor.Id);
            inserir.Parameters.AddWithValue("@v", e.Livro.Id);
            inserir.Parameters.AddWithValue("@de", e.DataEmprestimo);
            inserir.Parameters.AddWithValue("@dp", e.DataPrevista);
            inserir.ExecuteNonQuery();

            using var baixar = new MySqlCommand(
                "UPDATE livros SET disponiveis = disponiveis - 1 " +
                "WHERE id = @v AND disponiveis > 0", con, tx);   // "AND disponiveis > 0" protege se alguém pegou o último exemplar agora
            baixar.Parameters.AddWithValue("@v", e.Livro.Id);
            int afetadas = baixar.ExecuteNonQuery();             // devolve quantas linhas foram alteradas
            if (afetadas != 1)                                   // 0 = não havia mais exemplar
                throw new InvalidOperationException("O último exemplar acabou de ser emprestado.");

            tx.Commit();                                         // tudo certo: grava definitivamente as duas mudanças
        }
        catch
        {
            tx.Rollback();                                       // algo falhou: desfaz as duas mudanças
            throw;                                               // relança o erro para a tela tratar
        }
    }

    public void RegistrarDevolucao(Emprestimo e, DateTime dataDevolucao, decimal multa) // RF07
    {
        using var con = Conexao.Criar();
        con.Open();
        using var tx = con.BeginTransaction();
        try
        {
            using var devolver = new MySqlCommand(
                "UPDATE emprestimos SET data_devolucao = @d, multa = @m " +
                "WHERE id = @id AND data_devolucao IS NULL", con, tx); // só devolve o que ainda está ativo
            devolver.Parameters.AddWithValue("@d", dataDevolucao.Date);
            devolver.Parameters.AddWithValue("@m", multa);
            devolver.Parameters.AddWithValue("@id", e.Id);
            if (devolver.ExecuteNonQuery() != 1)
                throw new InvalidOperationException("Este empréstimo já foi devolvido.");

            using var repor = new MySqlCommand(
                "UPDATE livros SET disponiveis = disponiveis + 1 WHERE id = @v", con, tx);
            repor.Parameters.AddWithValue("@v", e.Livro.Id);
            repor.ExecuteNonQuery();

            tx.Commit();
        }
        catch
        {
            tx.Rollback();
            throw;
        }
    }
}

Sprint 3 · Programe a tela

Dê clique duplo no formulário (Load) e em cada botão para criar os eventos. A tela só orquestra: pede dados ao DAO, entrega ao serviço, manda o DAO gravar e recarrega. Veja como o tratamento de erros separa regra de negócio (aviso amarelo) de falha técnica (erro vermelho).

using Biblioteca.Core;
using Biblioteca.Data;
using MySqlConnector;

namespace Biblioteca.UI;

public partial class FrmEmprestimos : Form
{
    private readonly LeitorDAO leitorDAO = new();                // acesso a dados
    private readonly LivroDAO livroDAO = new();
    private readonly EmprestimoDAO emprestimoDAO = new();
    private readonly BibliotecaService servico = new();          // regras de negócio
    private List<Emprestimo> ativos = new();                     // cópia local dos empréstimos ativos (para achar o selecionado)

    public FrmEmprestimos()
    {
        InitializeComponent();                                   // monta os controles do designer
    }

    private void FrmEmprestimos_Load(object sender, EventArgs e)
    {
        Executar(Recarregar);                                    // carrega tudo dentro do tratamento de erros
        tslStatus.Text = "Pronto.";
    }

    // Executar roda uma ação com tratamento de erros em UM só lugar (reutilização).
    private void Executar(Action acao)
    {
        try
        {
            acao();                                              // roda o método recebido
        }
        catch (InvalidOperationException ex)                     // violação de REGRA DE NEGÓCIO (vem do serviço ou do DAO)
        {
            tslStatus.Text = ex.Message;                         // mensagem clara: o que houve
            MessageBox.Show(ex.Message, "Não foi possível concluir",
                MessageBoxButtons.OK, MessageBoxIcon.Warning);   // ícone de aviso: o usuário pode corrigir
        }
        catch (MySqlException)                                   // problema TÉCNICO de banco
        {
            const string msg = "Não foi possível acessar o banco. Verifique se o MySQL do Laragon está iniciado.";
            tslStatus.Text = msg;
            MessageBox.Show(msg, "Erro no banco de dados",
                MessageBoxButtons.OK, MessageBoxIcon.Error);     // ícone de erro: não depende do usuário
        }
    }

    private void Recarregar()                                    // lê o banco e atualiza a tela (fonte da verdade)
    {
        cmbLeitor.DataSource = leitorDAO.Listar();               // lista de Leitor (Aluno e Professor misturados)

        cmbLivro.DataSource = livroDAO.Listar("")                // "" = todos os livros
            .Where(l => l.TemDisponivel)                         // só os que têm exemplar (prevenção de erro)
            .ToList();

        ativos = emprestimoDAO.ListarAtivos();                   // empréstimos não devolvidos
        DateTime hoje = DateTime.Today;                          // data de hoje (sem hora)

        dgvAtivos.DataSource = ativos                            // projeta só o que a grade precisa mostrar
            .Select(x => new
            {
                x.Id,                                            // usado para achar o empréstimo selecionado
                Leitor = x.Leitor.Nome,
                Livro = x.Livro.Titulo,
                Retirada = x.DataEmprestimo.ToString("dd/MM/yyyy"),
                Prevista = x.DataPrevista.ToString("dd/MM/yyyy"),
                Situação = x.EstaAtrasado(hoje) ? "ATRASADO" : "No prazo"
            })
            .ToList();
        dgvAtivos.ClearSelection();                              // nenhuma linha selecionada ao carregar
    }

    private void btnEmprestar_Click(object sender, EventArgs e)
    {
        var leitor = cmbLeitor.SelectedItem as Leitor;           // "as" converte; null se não houver seleção
        var livro = cmbLivro.SelectedItem as Livro;
        if (leitor == null || livro == null)                     // validação básica de preenchimento
        {
            tslStatus.Text = "Escolha o leitor e o livro.";
            return;
        }

        Executar(() =>
        {
            var doLeitor = emprestimoDAO.ListarAtivos()          // dados frescos do banco
                .Where(x => x.Leitor.Id == leitor.Id)            // só os empréstimos deste leitor
                .ToList();

            var emprestimo = servico.Emprestar(leitor, livro, doLeitor, DateTime.Today); // REGRAS: pode lançar exceção
            emprestimoDAO.Registrar(emprestimo);                 // PERSISTÊNCIA: só chega aqui se as regras permitiram

            tslStatus.Text = $"\"{livro.Titulo}\" emprestado a {leitor.Nome}. Devolver até {emprestimo.DataPrevista:dd/MM/yyyy}.";
            Recarregar();                                        // atualiza listas e grade
        });
    }

    private void btnDevolver_Click(object sender, EventArgs e)
    {
        if (dgvAtivos.CurrentRow == null)                        // nenhuma linha escolhida
        {
            tslStatus.Text = "Selecione um empréstimo na lista.";
            return;
        }

        int id = (int)dgvAtivos.CurrentRow.Cells["Id"].Value;    // lê o Id da linha selecionada
        var emprestimo = ativos.First(x => x.Id == id);          // acha o objeto completo na lista local

        Executar(() =>
        {
            DateTime hoje = DateTime.Today;
            decimal multa = servico.CalcularMultaDevolucao(emprestimo, hoje); // REGRA: só calcula
            emprestimoDAO.RegistrarDevolucao(emprestimo, hoje, multa);         // PERSISTÊNCIA atômica

            tslStatus.Text = multa > 0
                ? $"Devolvido com atraso. Multa: {multa:C2}."    // C2 = moeda com 2 casas
                : "Devolvido no prazo. Sem multa.";
            Recarregar();
        });
    }
}
