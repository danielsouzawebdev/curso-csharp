Projeto 1 · Classe, objeto e referência

Objetivo: criar sua primeira classe, instanciar objetos e entender que variáveis de classe guardam referências. Pergunta: por que p1.Nome mudou quando alteramos p3?

// ---------- Pessoa.cs ----------
public class Pessoa                                   // class: define o MOLDE de uma pessoa; public: visível em todo o projeto
{
    public string Nome { get; set; } = "";            // propriedade auto-implementada; = "" evita o valor nulo
    public int Idade { get; set; }                    // o C# cria o campo escondido que guarda o valor

    public void Apresentar()                          // MÉTODO: uma ação que o objeto sabe fazer
    {
        Console.WriteLine($"Olá, eu sou {Nome} e tenho {Idade} anos.");   // usa os dados do PRÓPRIO objeto
    }

    public bool EhMaiorDeIdade() => Idade >= 18;      // método de uma linha (=>) que devolve true ou false
}

// ---------- Program.cs ----------
Console.OutputEncoding = System.Text.Encoding.UTF8;

Pessoa p1 = new Pessoa();                             // new cria um OBJETO a partir da classe
p1.Nome = "Ana";                                      // altera a propriedade com o operador ponto
p1.Idade = 20;

Pessoa p2 = new Pessoa { Nome = "Bruno", Idade = 15 };   // inicializador de objeto: cria e preenche de uma vez

p1.Apresentar();                                      // cada objeto usa os SEUS dados
p2.Apresentar();
Console.WriteLine($"{p2.Nome} é maior de idade? {p2.EhMaiorDeIdade()}");

// ---------- REFERÊNCIA: o ponto mais importante do projeto ----------
Pessoa p3 = p1;                                       // NÃO cria outra pessoa: copia a REFERÊNCIA (o "endereço")
p3.Nome = "Ana Maria";                                // altera o objeto para quem aponta para ele
Console.WriteLine($"p1.Nome = {p1.Nome}");            // mostra "Ana Maria": p1 e p3 são o MESMO objeto
Console.WriteLine($"p1 e p3 são o mesmo objeto? {ReferenceEquals(p1, p3)}");   // True
Console.WriteLine($"p1 e p2 são o mesmo objeto? {ReferenceEquals(p1, p2)}");   // False

Pessoa? ninguem = null;                               // Pessoa? permite null: "não aponta para objeto nenhum"
Console.WriteLine(ninguem?.Nome ?? "(ninguém)");      // ?. só acessa se não for null; ?? dá um valor alternativo
// Console.WriteLine(ninguem.Nome);                   // DESCOMENTE para ver a NullReferenceException

Projeto 2 · Encapsulamento (Produto)

Objetivo: proteger os dados com campo privado, propriedade com validação, construtor e métodos de regra. Experimentos: descomente a linha de ERRO e leia o código CS0200; tente criar o produto com preço zero; deixe o estoque baixo e veja o aviso no ToString.

// ---------- Produto.cs ----------
public class Produto
{
    private decimal preco;                            // CAMPO PRIVADO: só a classe enxerga
    private int quantidade;

    public string Nome { get; }                       // só get: definido no construtor e nunca mais muda

    public decimal Preco                              // propriedade com VALIDAÇÃO: a única porta de acesso ao preço
    {
        get => preco;                                 // leitura livre
        set
        {
            if (value <= 0)                           // "value" = valor que tentaram atribuir
                throw new ArgumentException("O preço deve ser maior que zero.");   // recusa o valor inválido
            preco = value;                            // só guarda se for válido
        }
    }

    public int Quantidade => quantidade;              // lê de fora, mas NÃO dá para alterar de fora (sem set)
    public bool EstoqueBaixo => quantidade < 5;       // propriedade CALCULADA: não guarda nada
    public decimal ValorEmEstoque => preco * quantidade;

    public Produto(string nome, decimal preco, int quantidade = 0)   // CONSTRUTOR; quantidade é opcional (padrão 0)
    {
        if (string.IsNullOrWhiteSpace(nome))
            throw new ArgumentException("O nome é obrigatório.");
        if (quantidade < 0)
            throw new ArgumentException("A quantidade não pode ser negativa.");

        Nome = nome.Trim();                           // guarda o nome sem espaços nas pontas
        Preco = preco;                                // usa a PROPRIEDADE (e portanto a validação do set)
        this.quantidade = quantidade;                 // this.quantidade = o campo; quantidade = o parâmetro (mesmo nome)
    }

    public void Entrada(int qtd)                      // regra de negócio dentro da classe
    {
        if (qtd <= 0)
            throw new ArgumentException("A quantidade de entrada deve ser maior que zero.");
        quantidade += qtd;
    }

    public bool Saida(int qtd)                        // devolve false se não puder retirar
    {
        if (qtd <= 0 || qtd > quantidade) return false;   // não deixa o estoque ficar negativo
        quantidade -= qtd;
        return true;
    }

    public override string ToString()                 // define como o objeto vira texto
        => $"{Nome,-15} {preco,10:C2}  {quantidade,4} un.{(EstoqueBaixo ? "  (ESTOQUE BAIXO)" : "")}";
}

// ---------- Program.cs ----------
Console.OutputEncoding = System.Text.Encoding.UTF8;

var arroz = new Produto("Arroz 5kg", 28.90m, 10);     // var: o compilador deduz o tipo (Produto)
Console.WriteLine(arroz);                             // WriteLine usa o ToString() sobrescrito

Console.WriteLine($"Retirar 4: {arroz.Saida(4)}");    // True
Console.WriteLine($"Retirar 50: {arroz.Saida(50)}");  // False: a classe PROTEGEU o estoque
Console.WriteLine(arroz);                             // agora com 6 un.

arroz.Saida(2);                                       // fica com 4 unidades
Console.WriteLine($"Estoque baixo? {arroz.EstoqueBaixo}");   // True

// arroz.Quantidade = 999;                            // DESCOMENTE: erro de compilação CS0200 (propriedade só de leitura)

try
{
    arroz.Preco = -5;                                 // o set da propriedade lança exceção
}
catch (ArgumentException ex)                          // captura o erro de validação
{
    Console.WriteLine($"Preço recusado: {ex.Message}");
}

try
{
    var invalido = new Produto("", 10);               // o construtor recusa nome vazio
}
catch (ArgumentException ex)
{
    Console.WriteLine($"Produto recusado: {ex.Message}");
}

Projeto 3 · Construtores, sobrecarga e static (Livro)

Objetivo: várias formas de criar o mesmo objeto, construtor encadeado, init e membros estáticos. Pergunta: qual é a diferença prática entre Id e TotalCriados? Por que proximoId é static?

// ---------- Livro.cs ----------
public class Livro
{
    private static int proximoId = 1;                 // STATIC: UMA cópia compartilhada por TODOS os livros

    public static int TotalCriados { get; private set; }   // estática: pertence à CLASSE, não ao objeto

    public int Id { get; }                            // cada livro tem o seu Id (membro de instância)
    public string Titulo { get; }
    public string Autor { get; }
    public int Paginas { get; }
    public string Editora { get; init; } = "Não informada";   // init: só pode ser definida na CRIAÇÃO do objeto

    public Livro(string titulo, string autor, int paginas)    // construtor principal (o mais completo)
    {
        Id = proximoId++;                             // usa o valor atual e depois soma 1: Id sequencial automático
        Titulo = titulo;
        Autor = autor;
        Paginas = paginas;
        TotalCriados++;                               // atualiza o contador da classe
    }

    public Livro(string titulo, string autor) : this(titulo, autor, 0)   // SOBRECARGA; ": this(...)" reaproveita o principal
    {
    }

    public Livro(string titulo) : this(titulo, "Autor desconhecido")     // encadeia no construtor anterior
    {
    }

    public static bool PaginasValidas(int paginas) => paginas > 0;       // método ESTÁTICO: chamado pela classe

    public string Resumo()
        => $"#{Id} {Titulo} - {Autor} ({(Paginas > 0 ? Paginas + " págs." : "págs. n/d")}) - {Editora}";
}

Console.OutputEncoding = System.Text.Encoding.UTF8;

var l3 = new Livro("Livro sem autor");                                  // 1 argumento: usa o 3º
var l4 = new Livro("Capitães da Areia", "Jorge Amado", 280) { Editora = "Companhia das Letras" };   // init na criação

Console.WriteLine(l4.Resumo());

Console.WriteLine($"\nLivros criados: {Livro.TotalCriados}");          // membro estático: Classe.Membro
Console.WriteLine($"Páginas válidas (0)? {Livro.PaginasValidas(0)}");   // False

Projeto 4 · Herança (Veículos)

Objetivo: reaproveitar código com herança, protected, base, virtual/override e um primeiro contato com polimorfismo. Experimentos: remova a palavra virtual da base e veja o erro CS0506; adicione uma classe Caminhao (com capacidade de carga e diária de 300).

// ---------- Veiculo.cs ----------
public class Veiculo                                  // CLASSE BASE
{
    protected double velocidadeAtual;                 // protected: a classe e as FILHAS enxergam; os de fora não

    public string Placa { get; }
    public string Modelo { get; }
    public int Ano { get; }
    public double VelocidadeAtual => velocidadeAtual; // leitura pública

    public Veiculo(string placa, string modelo, int ano)
    {
        Placa = placa;
        Modelo = modelo;
        Ano = ano;
    }

    public void Acelerar(double incremento) => velocidadeAtual += incremento;

    public virtual double CalcularDiaria() => 100;    // virtual: as filhas PODEM sobrescrever

    public virtual string Descrever() => $"{Modelo} ({Placa}), {Ano}";
}

// ---------- Carro.cs ----------
public class Carro : Veiculo                          // ": Veiculo" = Carro HERDA de Veiculo
{
    public int Portas { get; }

    public Carro(string placa, string modelo, int ano, int portas)
        : base(placa, modelo, ano)                    // base(...) chama o construtor da classe base
    {
        Portas = portas;
    }

    public void Frear(double reducao)                 // usa o campo protected da base
        => velocidadeAtual = Math.Max(0, velocidadeAtual - reducao);   // Math.Max evita velocidade negativa

    public override double CalcularDiaria()           // override: troca o comportamento herdado
        => base.CalcularDiaria() + 50;                // base.Metodo() reaproveita o cálculo da base (100 + 50)

    public override string Descrever()
        => base.Descrever() + $" | {Portas} portas";  // acrescenta ao que a base já faz
}

// ---------- Moto.cs ----------
public class Moto : Veiculo
{
    public int Cilindradas { get; }

    public Moto(string placa, string modelo, int ano, int cilindradas)
        : base(placa, modelo, ano)
    {
        Cilindradas = cilindradas;
    }

    public override double CalcularDiaria() => 70;    // moto é mais barata: ignora a regra da base

    public override string Descrever() => base.Descrever() + $" | {Cilindradas} cc";
}

// ---------- Program.cs ----------
Console.OutputEncoding = System.Text.Encoding.UTF8;

List<Veiculo> frota = new()                           // lista do tipo BASE guardando objetos de tipos DIFERENTES
{

    new Veiculo("GHI-9012", "Veículo genérico", 2020)

foreach (Veiculo v in frota)                          // a variável é Veiculo, mas cada objeto tem seu tipo real
{

    if (v is Carro c)                                 // "is" confere o tipo real e já cria a variável c
    {
        Console.WriteLine($"   (é um carro de {c.Portas} portas)");
    }
}

Projeto 5 · Classe abstrata e polimorfismo (Folha de pagamento)

Objetivo: definir um modelo com classe abstrata e deixar cada filha calcular do seu jeito. Experimentos: descomente o new Funcionario(...) (erro CS0144); apague o CalcularSalario de Mensalista (erro CS0534); crie Estagiario (bolsa fixa e imposto zero) sem alterar nenhuma outra classe.

Desenhe o diagrama de classes deste projeto (compare com o da Teoria).

// ---------- Funcionario.cs ----------
public abstract class Funcionario                     // abstract: NÃO dá para fazer "new Funcionario()"
{
    public string Nome { get; }

    protected Funcionario(string nome)                // construtor protected: só as filhas chamam
    {
        Nome = nome;
    }

    public abstract decimal CalcularSalario();        // abstract: SEM corpo; toda filha É OBRIGADA a implementar

    public virtual decimal CalcularImposto()          // virtual: tem comportamento padrão (10%), mas pode mudar
        => CalcularSalario() * 0.10m;                 // chama o método abstrato: usa a versão da filha!

    public override string ToString()
        => $"{GetType().Name,-13} {Nome,-10} Salário {CalcularSalario(),10:C2} | Imposto {CalcularImposto(),9:C2}";   // GetType().Name = nome da classe real
}

// ---------- Mensalista.cs ----------
public class Mensalista : Funcionario
{
    public decimal SalarioBase { get; }

    public Mensalista(string nome, decimal salarioBase) : base(nome)
    {
        SalarioBase = salarioBase;
    }

    public override decimal CalcularSalario() => SalarioBase;   // implementação obrigatória
}

// ---------- Horista.cs ----------
public class Horista : Funcionario
{
    public int HorasTrabalhadas { get; }
    public decimal ValorHora { get; }

    public Horista(string nome, int horas, decimal valorHora) : base(nome)
    {
        HorasTrabalhadas = horas;
        ValorHora = valorHora;
    }

    public override decimal CalcularSalario()
    {
        const int JornadaMensal = 160;                // horas normais do mês
        if (HorasTrabalhadas <= JornadaMensal)
            return HorasTrabalhadas * ValorHora;      // sem horas extras

        int extras = HorasTrabalhadas - JornadaMensal;                      // horas acima da jornada
        return JornadaMensal * ValorHora + extras * ValorHora * 1.5m;       // extras valem 50% a mais
    }
}

// ---------- Comissionado.cs ----------
public class Comissionado : Funcionario
{
    public decimal Fixo { get; }
    public decimal Vendas { get; }
    public decimal Percentual { get; }                // ex.: 5 significa 5%

    public Comissionado(string nome, decimal fixo, decimal vendas, decimal percentual) : base(nome)
    {
        Fixo = fixo;
        Vendas = vendas;
        Percentual = percentual;
    }

    public override decimal CalcularSalario() => Fixo + Vendas * Percentual / 100;

    public override decimal CalcularImposto() => CalcularSalario() * 0.15m;   // sobrescreve a regra padrão: 15%
}

// ---------- Program.cs ----------
Console.OutputEncoding = System.Text.Encoding.UTF8;

// Funcionario f = new Funcionario("X");              // DESCOMENTE: ERRO CS0144 (classe abstrata não se instancia)

List<Funcionario> folha = new()                       // POLIMORFISMO: uma lista, três tipos de funcionário
{
    new Mensalista("Ana", 4500m),
    new Horista("Bruno", 180, 30m),                   // 160 normais + 20 extras
    new Comissionado("Carla", 2000m, 50000m, 5m)      // fixo + 5% de 50 mil
};

foreach (var f in folha)
{
    Console.WriteLine(f);                             // ToString() usa os cálculos de cada tipo
}

decimal totalSalarios = folha.Sum(f => f.CalcularSalario());          // LINQ: soma usando o método polimórfico
decimal totalImpostos = folha.Sum(f => f.CalcularImposto());
Console.WriteLine($"\nTotal de salários: {totalSalarios:C2} | Total de impostos: {totalImpostos:C2}");

var maisBemPago = folha.OrderByDescending(f => f.CalcularSalario()).First();   // ordena do maior para o menor e pega o 1º
Console.WriteLine($"Maior salário: {maisBemPago.Nome}");

Projeto 6 · Interfaces (Notificadores e Ranking)

Objetivo: contrato com interface, baixo acoplamento e a interface IComparable<T> do próprio .NET. Experimento: crie NotificadorWhatsApp e use-o no ServicoPedidos sem alterá-lo. Pergunta: por que o serviço recebe INotificador no construtor, e não cria o notificador dentro dele?

// ---------- INotificador.cs ----------
public interface INotificador                         // INTERFACE: apenas o CONTRATO (o quê), sem implementação (como)
{
    void Enviar(string destinatario, string mensagem);
}

// ---------- NotificadorEmail.cs ----------
public class NotificadorEmail : INotificador          // ": INotificador" = implementa o contrato
{
    public void Enviar(string destinatario, string mensagem)
        => Console.WriteLine($"[E-MAIL para {destinatario}] {mensagem}");
}

// ---------- NotificadorSms.cs ----------
public class NotificadorSms : INotificador
{
    public void Enviar(string destinatario, string mensagem)
        => Console.WriteLine($"[SMS para {destinatario}] {mensagem.Substring(0, Math.Min(40, mensagem.Length))}...");   // SMS corta o texto
}

// ---------- ServicoPedidos.cs ----------
public class ServicoPedidos
{
    private readonly INotificador notificador;        // depende da INTERFACE, e não de uma classe concreta (baixo acoplamento)

    public ServicoPedidos(INotificador notificador)   // quem cria o serviço ESCOLHE o notificador ("injeção de dependência")
    {
        this.notificador = notificador;
    }

    public void Concluir(string cliente, int numeroPedido)
    {
        Console.WriteLine($"Pedido {numeroPedido} concluído.");              // regra do serviço
        notificador.Enviar(cliente, $"Seu pedido {numeroPedido} foi concluído e já está a caminho!");   // não sabe se é e-mail ou SMS
    }
}

// ---------- Aluno.cs (interface do .NET: IComparable<T>) ----------
public class Aluno : IComparable<Aluno>               // permite ORDENAR objetos Aluno com Sort()
{
    public string Nome { get; }
    public double Media { get; }

    public Aluno(string nome, double media)
    {
        Nome = nome;
        Media = media;
    }

    public int CompareTo(Aluno? outro)                // contrato da interface: <0 antes, 0 igual, >0 depois
    {
        if (outro is null) return 1;                  // qualquer aluno vem depois de "nada"
        return outro.Media.CompareTo(Media);          // ordem DECRESCENTE de média (maior primeiro)
    }

    public override string ToString() => $"{Nome} ({Media:F1})";
}

// ---------- Program.cs ----------
Console.OutputEncoding = System.Text.Encoding.UTF8;

INotificador porEmail = new NotificadorEmail();       // variável do tipo da INTERFACE
INotificador porSms = new NotificadorSms();

new ServicoPedidos(porEmail).Concluir("ana@exemplo.com", 101);   // mesmo serviço, comportamento diferente
new ServicoPedidos(porSms).Concluir("(11) 91234-5678", 102);     // sem alterar uma linha de ServicoPedidos

var turma = new List<Aluno>
{
    new Aluno("Bruno", 7.5), new Aluno("Ana", 9.0), new Aluno("Carla", 6.0)
};
turma.Sort();                                         // Sort usa o CompareTo que implementamos
Console.WriteLine("\nRanking: " + string.Join(", ", turma));   // Ana (9,0), Bruno (7,5), Carla (6,0)

Projeto 7 · Composição e coleções (Pedido)

Objetivo: objetos que contêm outros, lista privada exposta como somente leitura e LINQ. Pergunta: por que AdicionarItem pede ao Produto para dar a saída, em vez de verificar o estoque no próprio Pedido? Desafio: adicione RemoverItem, que devolve a quantidade ao estoque (Entrada).

// ---------- Cliente.cs ----------
public class Cliente
{
    public string Nome { get; }
    public string Email { get; }

    public Cliente(string nome, string email)
    {
        if (string.IsNullOrWhiteSpace(nome)) throw new ArgumentException("Nome obrigatório.");
        Nome = nome.Trim();
        Email = email.Trim();
    }
}

// ---------- ItemPedido.cs ----------
public class ItemPedido
{
    public Produto Produto { get; }                   // COMPOSIÇÃO/ASSOCIAÇÃO: um item "tem um" produto
    public int Quantidade { get; private set; }       // só a classe altera

    public decimal Subtotal => Produto.Preco * Quantidade;   // calculada: usa o objeto Produto

    public ItemPedido(Produto produto, int quantidade)
    {
        Produto = produto;
        Quantidade = quantidade;
    }

    public void AumentarQuantidade(int qtd) => Quantidade += qtd;
}

// ---------- Pedido.cs ----------
public class Pedido
{
    private readonly List<ItemPedido> itens = new();  // lista PRIVADA: ninguém de fora mexe nela diretamente

    public Cliente Cliente { get; }                   // um pedido "tem um" cliente
    public IReadOnlyList<ItemPedido> Itens => itens;  // visão SOMENTE LEITURA da lista
    public decimal Total => itens.Sum(i => i.Subtotal);   // LINQ: soma os subtotais

    public Pedido(Cliente cliente)
    {
        Cliente = cliente ?? throw new ArgumentNullException(nameof(cliente));   // ?? throw: recusa cliente nulo
    }

    public void AdicionarItem(Produto produto, int quantidade)
    {
        if (!produto.Saida(quantidade))               // pede ao Produto para baixar o estoque ("diga, não pergunte")
            throw new InvalidOperationException($"Estoque insuficiente de {produto.Nome}.");

        var existente = itens.FirstOrDefault(i => i.Produto == produto);   // procura o mesmo produto no pedido
        if (existente != null)
            existente.AumentarQuantidade(quantidade); // já existe: só aumenta a quantidade
        else
            itens.Add(new ItemPedido(produto, quantidade));   // não existe: cria um novo item
    }

    public void Imprimir()
    {
        Console.WriteLine($"\nPedido de {Cliente.Nome} <{Cliente.Email}>");
        foreach (var item in itens)
            Console.WriteLine($"  {item.Produto.Nome,-15} {item.Quantidade,3} x {item.Produto.Preco,9:C2} = {item.Subtotal,10:C2}");
        Console.WriteLine($"  {"TOTAL",-15} {Total,29:C2}");
    }
}

// ---------- Program.cs (copie Produto.cs do Projeto 2 para este projeto) ----------
Console.OutputEncoding = System.Text.Encoding.UTF8;

var arroz = new Produto("Arroz 5kg", 28.90m, 10);
var feijao = new Produto("Feijão 1kg", 8.50m, 20);
var ana = new Cliente("Ana Souza", "ana@exemplo.com");

var pedido = new Pedido(ana);
pedido.AdicionarItem(arroz, 2);
pedido.AdicionarItem(feijao, 3);

Caça aos defeitos de design

Esta classe compila (quase), mas tem problemas de design e de tipos. Identifique e corrija usando o que aprendeu:

public class ContaPoupanca
{
    public decimal Saldo;                            // Defeito 1?

    public ContaPoupanca(decimal saldo)
    {
        Saldo = saldo;
    }

    public void Sacar(decimal valor)
    {
        Saldo -= valor;                              // Defeito 2?
    }

    public void AplicarRendimento()
    {
        Saldo = Saldo + Saldo * 0.005;               // Defeito 3? (compile e leia o erro)
    }
}
Ver os defeitos
  1. Campo público: qualquer um altera o saldo. Use public decimal Saldo { get; private set; }.
  2. Sem validação no saque: permite sacar mais que o saldo e valores negativos. Valide e lance exceção (InvalidOperationException para saldo insuficiente, ArgumentException para valor inválido).
  3. Erro de tipo (CS0266/CS0019): 0.005 é double e não pode se misturar com decimal. Use 0.005m.
  4. Extra: o construtor aceita saldo inicial negativo; valide.

8.1 · Políticas de desconto

// ---------- IDesconto.cs ----------
public interface IDesconto                            // contrato: toda política de desconto calcula um VALOR a descontar
{
    string Nome { get; }
    decimal Calcular(decimal total);
}

public class SemDesconto : IDesconto
{
    public string Nome => "Sem desconto";
    public decimal Calcular(decimal total) => 0m;
}

public class DescontoPercentual : IDesconto
{
    private readonly decimal percentual;

    public string Nome => $"{percentual}% de desconto";

    public DescontoPercentual(decimal percentual)
    {
        if (percentual <= 0 || percentual > 50)       // regra de negócio: até 50%
            throw new ArgumentException("O percentual deve estar entre 0 e 50.");
        this.percentual = percentual;
    }

    public decimal Calcular(decimal total) => total * percentual / 100;
}

public class DescontoFixo : IDesconto
{
    private readonly decimal valor;

    public string Nome => $"Desconto de {valor:C2}";

    public DescontoFixo(decimal valor)
    {
        if (valor <= 0) throw new ArgumentException("O valor deve ser maior que zero.");
        this.valor = valor;
    }

    public decimal Calcular(decimal total) => Math.Min(valor, total);   // nunca desconta mais que o total
}

Projeto 8 · Pedido com desconto e a classe Loja

// ---------- Pedido.cs: acrescente estes membros à classe Pedido do Projeto 7 ----------
public IDesconto Desconto { get; set; } = new SemDesconto();            // por padrão, sem desconto

public decimal ValorDesconto => Desconto.Calcular(Total);               // POLIMORFISMO: cada política calcula do seu jeito
public decimal TotalComDesconto => Total - ValorDesconto;

// No método Imprimir(), antes do TOTAL, acrescente:
//   Console.WriteLine($"  {Desconto.Nome,-15} {-ValorDesconto,29:C2}");
//   e troque {Total} por {TotalComDesconto} na linha do TOTAL.

// ---------- Loja.cs ----------
public class Loja                                     // "serviço": organiza os dados e as regras do sistema
{
    private readonly List<Produto> catalogo = new();
    private readonly List<Cliente> clientes = new();
    private readonly List<Pedido> pedidos = new();

    public IReadOnlyList<Produto> Catalogo => catalogo;
    public IReadOnlyList<Cliente> Clientes => clientes;
    public IReadOnlyList<Pedido> Pedidos => pedidos;

    public void CadastrarProduto(string nome, decimal preco, int quantidade)
    {
        if (catalogo.Any(p => p.Nome.Equals(nome.Trim(), StringComparison.OrdinalIgnoreCase)))   // sem diferenciar maiúsculas
            throw new InvalidOperationException("Já existe um produto com esse nome.");
        catalogo.Add(new Produto(nome, preco, quantidade));
    }

    public void CadastrarCliente(string nome, string email)
    {
        if (clientes.Any(c => c.Email.Equals(email.Trim(), StringComparison.OrdinalIgnoreCase)))
            throw new InvalidOperationException("Já existe um cliente com esse e-mail.");
        clientes.Add(new Cliente(nome, email));
    }

    public Pedido NovoPedido(Cliente cliente)
    {
        var pedido = new Pedido(cliente);
        pedidos.Add(pedido);                          // a Loja guarda todos os pedidos
        return pedido;
    }

    public decimal Faturamento() => pedidos.Sum(p => p.TotalComDesconto);

    public IEnumerable<Produto> ProdutosComEstoqueBaixo() => catalogo.Where(p => p.EstoqueBaixo);
}

Projeto 8 · Menu do sistema

O menu usa funções auxiliares do Módulo 1 (LerInteiro etc.) e um único try/catch para transformar as exceções das regras em mensagens ao usuário. Teste: produto repetido, estoque insuficiente, desconto de 60%, cliente sem pedidos.

// ---------- Program.cs ----------
Console.OutputEncoding = System.Text.Encoding.UTF8;

var loja = new Loja();
loja.CadastrarProduto("Arroz 5kg", 28.90m, 10);       // dados iniciais para testar
loja.CadastrarProduto("Feijão 1kg", 8.50m, 20);
loja.CadastrarCliente("Ana Souza", "ana@exemplo.com");

int opcao;
do
{
    Console.WriteLine("\n=== MINI LOJA ===");
    Console.WriteLine("1 - Cadastrar produto   2 - Cadastrar cliente   3 - Novo pedido");
    Console.WriteLine("4 - Listar produtos     5 - Relatório           0 - Sair");
    opcao = LerInteiro("Escolha: ");

    try                                               // UM try/catch para todo o menu: regras e validações viram mensagens
    {
        switch (opcao)
        {
            case 1: CadastrarProduto(loja); break;
            case 2: CadastrarCliente(loja); break;
            case 3: NovoPedido(loja); break;
            case 4: foreach (var p in loja.Catalogo) Console.WriteLine(p); break;
            case 5: Relatorio(loja); break;
            case 0: Console.WriteLine("Até logo!"); break;
            default: Console.WriteLine("Opção inválida."); break;
        }
    }
    catch (Exception ex) when (ex is ArgumentException or InvalidOperationException)   // só as exceções de regra
    {
        Console.WriteLine($"Atenção: {ex.Message}");
    }
} while (opcao != 0);

static int LerInteiro(string msg)
{
    Console.Write(msg);
    int v;
    while (!int.TryParse(Console.ReadLine(), out v)) Console.Write("Inválido. " + msg);
    return v;
}

static decimal LerDecimal(string msg)
{
    Console.Write(msg);
    decimal v;
    while (!decimal.TryParse(Console.ReadLine(), out v)) Console.Write("Inválido. " + msg);
    return v;
}

static string LerTexto(string msg)
{
    Console.Write(msg);
    return (Console.ReadLine() ?? "").Trim();
}

static void CadastrarProduto(Loja loja)
{
    string nome = LerTexto("Nome: ");
    decimal preco = LerDecimal("Preço: ");
    int qtd = LerInteiro("Quantidade inicial: ");
    loja.CadastrarProduto(nome, preco, qtd);          // a Loja e o Produto validam: o menu só pede os dados
    Console.WriteLine("Produto cadastrado!");
}

static void CadastrarCliente(Loja loja)
{
    loja.CadastrarCliente(LerTexto("Nome: "), LerTexto("E-mail: "));
    Console.WriteLine("Cliente cadastrado!");
}

static void NovoPedido(Loja loja)
{
    if (loja.Clientes.Count == 0 || loja.Catalogo.Count == 0)
    {
        Console.WriteLine("Cadastre ao menos um cliente e um produto.");
        return;
    }

    for (int i = 0; i < loja.Clientes.Count; i++)
        Console.WriteLine($"{i + 1} - {loja.Clientes[i].Nome}");
    int c = LerInteiro("Cliente: ");
    if (c < 1 || c > loja.Clientes.Count) { Console.WriteLine("Cliente inválido."); return; }

    var pedido = loja.NovoPedido(loja.Clientes[c - 1]);

    while (true)                                      // laço infinito controlado por break
    {
        for (int i = 0; i < loja.Catalogo.Count; i++)
            Console.WriteLine($"{i + 1} - {loja.Catalogo[i]}");
        int p = LerInteiro("Produto (0 para finalizar): ");
        if (p == 0) break;                            // sai do laço
        if (p < 1 || p > loja.Catalogo.Count) { Console.WriteLine("Produto inválido."); continue; }

        int qtd = LerInteiro("Quantidade: ");
        try
        {
            pedido.AdicionarItem(loja.Catalogo[p - 1], qtd);
        }
        catch (InvalidOperationException ex)          // erro de um item não perde o pedido inteiro
        {
            Console.WriteLine($"Atenção: {ex.Message}");
        }
    }

    Console.WriteLine("Desconto: 1 - Nenhum   2 - Percentual   3 - Valor fixo");
    switch (LerInteiro("Escolha: "))
    {
        case 2: pedido.Desconto = new DescontoPercentual(LerDecimal("Percentual (até 50): ")); break;
        case 3: pedido.Desconto = new DescontoFixo(LerDecimal("Valor do desconto: ")); break;
    }                                                 // POLIMORFISMO: o pedido não sabe qual desconto recebeu
    pedido.Imprimir();
}

static void Relatorio(Loja loja)
{
    Console.WriteLine($"\nPedidos: {loja.Pedidos.Count} | Faturamento: {loja.Faturamento():C2}");
    Console.WriteLine("Produtos com estoque baixo:");
    foreach (var p in loja.ProdutosComEstoqueBaixo())
        Console.WriteLine($"  {p.Nome} ({p.Quantidade} un.)");
}
